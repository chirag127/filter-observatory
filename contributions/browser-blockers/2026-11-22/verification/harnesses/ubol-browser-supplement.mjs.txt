function inventory(directory, relative = '') {
    const out = {};
    for ( const entry of readdirSync(path.join(directory, relative), { withFileTypes: true }) ) {
        const name = path.posix.join(relative, entry.name);
        assert.equal(entry.isSymbolicLink(), false, `unexpected symbolic link: ${name}`);
        if ( entry.isDirectory() ) {
            Object.assign(out, inventory(directory, name));
        } else if ( entry.isFile() ) {
            out[name] = sha256(readFileSync(path.join(directory, name)));
        }
    }
    return out;
}

async function installProbe(page, mode) {
    await page.evaluate(mode => {
        const probe = {
            mode, readTerminal: false, storageRejected: false,
            reported: [], unhandled: [], nativeReadStarted: false,
        };
        globalThis.__ubolRecoveryProbe = probe;
        const originalError = console.error;
        console.error = (...args) => {
            probe.reported.push(args.map(a => String(a)).join(' '));
            originalError.apply(console, args);
        };
        addEventListener('unhandledrejection', event => {
            probe.unhandled.push(String(event.reason));
        });
        const NativeReader = globalThis.FileReader;
        globalThis.FileReader = class extends NativeReader {
            constructor() {
                super();
                this.addEventListener('loadend', () => { probe.readTerminal = true; });
            }
            readAsText(file, ...args) {
                if ( probe.mode === 'read-throw' ) {
                    probe.readTerminal = true;
                    throw new DOMException('Injected synchronous file read failure', 'SecurityError');
                }
                if ( probe.mode === 'read-error' ) {
                    queueMicrotask(() => {
                        Object.defineProperty(this, 'error', {
                            value: new DOMException('Injected file read failure', 'NotReadableError'),
                        });
                        Object.defineProperty(this, 'readyState', { value: this.DONE });
                        this.dispatchEvent(new ProgressEvent('error'));
                        this.dispatchEvent(new ProgressEvent('loadend'));
                    });
                    return;
                }
                probe.nativeReadStarted = true;
                super.readAsText(file, ...args);
                if ( probe.mode === 'read-abort' ) { this.abort(); }
            }
        };
        if ( mode === 'storage-error' ) {
            const originalRemove = chrome.storage.local.remove.bind(chrome.storage.local);
            chrome.storage.local.remove = keys => {
                const list = Array.isArray(keys) ? keys : [keys];
                if ( list.includes('userDnrRules') ) {
                    probe.storageRejected = true;
                    return Promise.reject(new Error('Injected extension-page storage rejection'));
                }
                return originalRemove(keys);
            };
        }
    }, mode);
}

async function observe(page) {
    return page.evaluate(async () => {
        await new Promise(resolve => requestAnimationFrame(() => resolve()));
        const current = await chrome.runtime.sendMessage({ what: 'getCurrentConfig' });
        return {
            busy: document.body.classList.contains('busy'),
            settings_display: getComputedStyle(document.querySelector('section[data-pane="settings"]')).display,
            autoReload: current?.autoReload,
            probe: globalThis.__ubolRecoveryProbe,
        };
    });
}

async function selectRestore(page, body, empty = false) {
    const chooserPromise = page.waitForEvent('filechooser', { timeout: 8000 });
    await page.locator('section[data-pane="settings"] button:has([data-i18n="restoreButton"])').click();
    const chooser = await chooserPromise;
    await chooser.setFiles(empty ? [] : {
        name: 'synthetic-ubol-settings.json', mimeType: 'application/json', buffer: Buffer.from(body),
    });
    return { native_filechooser_event: true, empty_selection: empty };
}

async function clickReset(page, accepted) {
    const dialogPromise = page.waitForEvent('dialog', { timeout: 8000 });
    const click = page.locator('section[data-pane="settings"] button:has([data-i18n="resetToDefaultButton"])').click();
    const dialog = await dialogPromise;
    assert.equal(dialog.type(), 'confirm');
    if ( accepted ) { await dialog.accept(); } else { await dialog.dismiss(); }
    await click;
    return { native_confirm_event: true, accepted };
}

async function runVariant(name, expectedFixed, distributionHashes) {
    const extension = path.join(work, `${name}-extension`);
    const profile = path.join(work, `${name}-profile`);

    const differences = Object.keys(distributionHashes).filter(file => copied[file] !== distributionHashes[file]);
    assert.deepEqual(differences, expectedFixed ? ['js/settings.js'] : []);
    assert.equal(Object.keys(copied).length, Object.keys(distributionHashes).length);

    const result = {
        name, expected_fixed: expectedFixed, extension_directory: extension, profile_directory: profile,

    save();
    let context;
    let browser;
    let closed = false;
    try {
        context = await chromium.launchPersistentContext(profile, {
            channel: 'chromium', executablePath: executable, headless: true, offline: true,
            viewport: { width: 1280, height: 960 }, reducedMotion: 'reduce', timeout: 30000,
            args: [`--disable-extensions-except=${extension}`, `--load-extension=${extension}`],
        });
        context.on('close', () => { closed = true; });
        browser = context.browser();
        result.browser_version = browser?.version();
        await context.route(/^https?:\/\//, route => route.abort('blockedbyclient'));
        const worker = context.serviceWorkers()[0] ?? await context.waitForEvent('serviceworker', { timeout: 15000 });
        result.extension_id = new URL(worker.url()).hostname;
        assert.match(worker.url(), /^chrome-extension:\/\/[^/]+\/js\/background\.js$/);
        const dashboardURL = `chrome-extension://${result.extension_id}/dashboard.html`;
        const cases = [
            { id: 'valid-restore', operation: 'restore', mode: 'native', text: '{"autoReload":false}', expectedAutoReload: false },
            { id: 'empty-file-selection', operation: 'restore', mode: 'native', empty: true, expectedAutoReload: true },
            { id: 'invalid-json', operation: 'restore', mode: 'native', text: '{not json', expectedAutoReload: true },
            { id: 'read-error', operation: 'restore', mode: 'read-error', failure: true, expectedAutoReload: true },
            { id: 'native-read-abort', operation: 'restore', mode: 'read-abort', failure: true, expectedAutoReload: true },
            { id: 'read-synchronous-throw', operation: 'restore', mode: 'read-throw', failure: true, expectedAutoReload: true },
            { id: 'restore-storage-rejection', operation: 'restore', mode: 'storage-error', failure: true, text: '{"autoReload":false}', expectedAutoReload: false },
            { id: 'confirmed-reset', operation: 'reset', mode: 'native', accept: true, initialAutoReload: false, expectedAutoReload: true },

        for ( const item of cases ) {
            const page = await context.newPage();
            const errors = [];
            page.on('pageerror', error => { errors.push({ name: error.name, message: error.message }); });
            const testResult = { id: item.id, started_at: new Date().toISOString() };
            result.cases.push(testResult);
            try {
                await page.goto(dashboardURL, { waitUntil: 'load', timeout: 15000 });
                await page.locator('body:not(.loading) section[data-pane="settings"]').waitFor({ state: 'visible', timeout: 15000 });
                await page.evaluate(async state => {
                    await chrome.runtime.sendMessage({ what: 'setAutoReload', state });
                }, item.initialAutoReload ?? true);
                await installProbe(page, item.mode);
                if ( item.operation === 'restore' ) {
                    testResult.interaction = await selectRestore(page, item.text ?? '{}', item.empty);
                    if ( !item.empty ) {
                        await page.waitForFunction(() => globalThis.__ubolRecoveryProbe.readTerminal, undefined, { timeout: 10000 });
                    }
                } else {
                    testResult.interaction = await clickReset(page, item.accept);
                }
                if ( item.mode === 'storage-error' ) {
                    await page.waitForFunction(() => {
                        const p = globalThis.__ubolRecoveryProbe;
                        return p.storageRejected && (p.reported.length !== 0 || p.unhandled.length !== 0);
                    }, undefined, { timeout: 10000 });
                }
                if ( !item.failure || expectedFixed ) {
                    await page.locator('body:not(.busy) section[data-pane="settings"]').waitFor({ state: 'visible', timeout: 10000 });
                }
                const observed = await observe(page);
                testResult.observed = observed;
                testResult.page_errors = errors;
                assert.equal(observed.busy, Boolean(item.failure && !expectedFixed), `${name}: ${item.id}: busy state`);
                assert.equal(observed.autoReload, item.expectedAutoReload, `${name}: ${item.id}: settings value`);
                if ( expectedFixed || !item.failure ) {
                    assert.equal(errors.length, 0, `${name}: ${item.id}: no uncaught page error`);
                    assert.equal(observed.probe.unhandled.length, 0, `${name}: ${item.id}: no unhandled rejection`);
                }
                if ( expectedFixed && item.failure && item.mode !== 'read-abort' ) {
                    assert.notEqual(observed.probe.reported.length, 0, `${item.id}: failure diagnostic`);
                }
                if ( ['read-error', 'confirmed-reset', 'valid-restore'].includes(item.id) ) {
                    testResult.screenshot = path.join(work, `${name}-${item.id}.png`);

                save();

        result.cleanup.context_close_event = closed;

        } catch ( error ) {
            result.cleanup.profile_removed = false;
            result.cleanup.error = { name: error.name, message: error.message };
        }

    }
    assert.equal(result.cleanup.context_close_event, true);

    assert.equal(result.cleanup.profile_removed, true);
